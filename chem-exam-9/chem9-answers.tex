% ============================ السؤال الثاني ============================
\question{السؤال الثاني: أكملي الفراغات الآتية}{10 علامات}

\begin{enumerate}[label=\textbf{\arabic*-}, itemsep=0.04cm, leftmargin=*]
\item يُعرّف الحمض بأنه: \ans{مادة تنتج أيونات الهيدروجين \en{\ce{H+}} عند ذوبانها في الماء.} \m{1}
\item من الأمثلة على الحموض القوية: \ans{حمض الهيدروكلوريك \en{\ce{HCl}}} \ (أو \en{\ce{HNO3}} ، \en{\ce{H2SO4}}) \m{1}
\item المادة التي يتغير لونها تبعًا لنوع المحلول: \ans{الكاشف} \ (مثل ورق تباع الشمس) \m{1}
\item تُعرّف القواعد بأنها: \ans{مواد تنتج أيونات الهيدروكسيد \en{\ce{OH-}} عند ذوبانها في الماء.} \m{1}
\item من الأمثلة على القواعد الضعيفة: \ans{الأمونيا \en{\ce{NH3}}} \m{1}
\item من خصائص الحموض: \ans{طعمها حامض} \ (أو: تحوّل لون ورق تباع الشمس الأزرق إلى الأحمر، أو تتفاعل مع الفلزات النشطة وينطلق غاز \en{\ce{H2}}) \m{1}
\item تُسمى المجموعة السابعة: \ans{الهالوجينات} \m{1} \par
ومن استخدامات عناصرها في صناعة: \ans{المبيّضات ومواد التعقيم} \ (أو: معجون الأسنان، البلاستيك) \m{1}
\item يُعرّف الملح بأنه: \ans{مركب أيوني ينتج من تفاعل الحمض مع القاعدة.} \m{1}
\item توصل محاليل الحموض التيار الكهربائي بسبب: \ans{وجود أيونات حرة الحركة في المحلول.} \m{1}
\end{enumerate}

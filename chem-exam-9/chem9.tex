\ce{_{18}Ar} \qquad\quad \ce{_{35}Br} \qquad\quad \ce{_{13}Al} \qquad\quad \ce{_{19}K}}}

\vspace{-0.2cm}

\begin{enumerate}[label=\textbf{\arabic*)}, itemsep=0.32cm, leftmargin=*]
\item اكتبي التوزيع الإلكتروني لكل عنصر:
\par\vspace{0.3cm}
\noindent\begin{tabularx}{\linewidth}{@{}X@{\hspace{0.8cm}}X@{}}
\en{\ce{_{18}Ar}} : \fillline & \en{\ce{_{35}Br}} : \fillline \\[0.45cm]
\en{\ce{_{13}Al}} : \fillline & \en{\ce{_{19}K}} : \fillline
\end{tabularx}

\item حددي العنصر الذي يقع في المجموعة الثالثة. \fillline

\item ما اسم مجموعة العنصر \en{\ce{K}}؟ وفي أي مجموعة يقع؟ \fillline

\item اكتبي رمز العنصر النبيل من بين هذه العناصر. \fillline

\item إلى أي دورة ينتمي العنصر \en{\ce{Br}}؟ وإلى أي مجموعة؟ \fillline

\item اكتبي التوزيع الإلكتروني للأيون \en{\ce{Al^{3+}}}. \fillline

\item أيهما أكبر حجمًا: \en{\ce{Ar}} أم \en{\ce{Al}}، علمًا بأنهما من نفس الدورة؟ \fillline

\item هل ينتمي العنصر \en{\ce{Br}} إلى الفلزات أم اللافلزات؟ \fillline

\item هل العنصر \en{\ce{Ar}} من العناصر الممثلة أم الانتقالية؟ \fillline
\end{enumerate}

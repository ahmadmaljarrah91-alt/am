\question{السؤال الأول: اختاري الإجابة الصحيحة لكل فقرة من الفقرات الآتية}{20 علامة}

\begin{enumerate}[label=\textbf{\arabic*-}, itemsep=0.03cm, leftmargin=*]

\item بالون مملوء بالغاز حجمه \en{30 L} عند درجة حرارة \en{40 °C} وضغط يساوي \en{1.5 atm}، فإن حجم البالون \en{(L)} في الظروف المعيارية يساوي:
\opts{\en{307}}{\en{22.4}}{\en{39.25}}{\en{30.7}}

\item السائل الذي له أقل طاقة تبخر مولية هو:
\opts{\chem{CH3CH2CH3}}{\chem{CH3CH2CH2CH3}}{\chem{CH3COCH3}}{\chem{CH3CH2OH}}

\item يبين المنحنى المجاور توزيع الطاقة الحركية على الجزيئات عند درجتي حرارة \en{$T_1$} و \en{$T_2$}. العبارة الصحيحة مما يأتي هي:
\par\vspace{0.1cm}
\noindent
\begin{minipage}[t]{0.6\linewidth}
\vspace{0pt}
\begin{tabularx}{\linewidth}{@{\hspace{0.5cm}}X@{}}
أ) درجة الحرارة \en{$T_2$} أكبر من \en{$T_1$}. \\
ب) عدد الجزيئات التي تمتلك طاقة كافية للتبخر عند \en{$T_1$} أكبر منها عند \en{$T_2$}. \\
جـ) بزيادة درجة الحرارة يقل الحد الأدنى من الطاقة اللازمة للتبخر. \\
د) متوسط الطاقة الحركية للجزيئات عند \en{$T_2$} أقل منها عند \en{$T_1$}.
\end{tabularx}
\end{minipage}\hfill
\begin{minipage}[t]{0.37\linewidth}
\vspace{0pt}\centering \kecurve
\end{minipage}

\item المادة الصلبة التي لها أعلى درجة انصهار:
\opts{\chem{AlF3}}{\chem{SiF4}}{\chem{NH3}}{\chem{P4}}

\item المادة الصلبة الجزيئية هي:
\opts{\chem{Al}}{\chem{BaCl2}}{\chem{Si}}{\chem{P4}}

\item الخصائص الآتية تزداد بزيادة تركيز مادة غير متطايرة في المحلول، ما عدا:
\optsTwo{الضغط الأسموزي.}{الارتفاع في درجة الغليان.}{الضغط البخاري.}{الانخفاض في درجة التجمد.}

\item حُضّر محلول مشبع من نترات البوتاسيوم \chem{KNO3} بإذابة \en{40 g} منه في \en{50 g} من الماء عند درجة حرارة \en{48 °C}، ثم بُرّد المحلول إلى \en{27 °C}، حيث ذائبية \chem{KNO3} عند هذه الدرجة تساوي \en{40 g/100 g H$_2$O}. فإن كتلة الملح المترسب \en{(g)} تساوي:
\opts{\en{40}}{صفر}{\en{20}}{\en{10}}

\item أحد الآتية يُعدّ محلولًا حقيقيًا:
\opts{الحليب}{السكر في الماء}{الطباشير في الماء}{التراب في الماء}

\item محلول مكوّن من \en{12.5 g} من مادة غير متطايرة مذابة في \en{170 g} من الماء، فأدّت إلى ارتفاع درجة غليان المحلول بمقدار \en{0.63 °C}
\en{($K_b$ H$_2$O = 0.512 °C kg mol$^{-1}$)}. فإن الكتلة المولية للمادة المذابة \en{(g/mol)} تساوي:
\opts{\en{88.5}}{\en{77.2}}{\en{59.5}}{\en{40.7}}

\item في التفاعل الآتي: \quad \chem{4NH3_{(g)} + 5O2_{(g)} <=> 4NO_{(g)} + 6H2O_{(g)}}
\par يؤدي سحب كمية من غاز \chem{NH3} من وعاء التفاعل إلى:
\optsTwo{زيادة سرعة التفاعل الأمامي.}{زيادة إنتاج غاز \chem{NO}.}{إزاحة موضع الاتزان نحو اليمين.}{إزاحة موضع الاتزان نحو اليسار.}

\item التفاعل الذي ينتج عنه كمية أكبر من النواتج عند زيادة الضغط المؤثِّر في وعاء التفاعل:
\optsTwo{\chem{I2_{(g)} + Br2_{(g)} <=> 2IBr_{(g)}}}{\chem{2NH3_{(g)} <=> N2_{(g)} + 3H2_{(g)}}}{\chem{2NO_{(g)} + Br2_{(g)} <=> 2NOBr_{(g)}}}{\chem{CH4_{(g)} + H2O_{(g)} <=> CO_{(g)} + 3H2_{(g)}}}

\item عند استعمال العامل المساعد في تفاعل إنتاج الأمونيا:
\optsTwo{يزداد تركيز الأمونيا عند الاتزان.}{يقل تركيز الأمونيا عند الاتزان.}{لا يتغير تركيز الأمونيا عند الاتزان.}{يزداد الزمن اللازم للوصول إلى موضع الاتزان.}

\item يتفكك غاز الفوسجين \chem{COCl2} وفق المعادلة: \quad \chem{COCl2_{(g)} <=> CO_{(g)} + Cl2_{(g)}}
\par فإذا كانت تراكيز المواد عند الاتزان: \quad
\LTRfig{$[\mathrm{CO}] = [\mathrm{Cl_2}] = 0.18\ \mathrm{M}$ \ , \ $[\mathrm{COCl_2}] = 0.75\ \mathrm{M}$} ،
فإن ثابت الاتزان للتفاعل يساوي:
\opts{\en{0.48}}{\en{0.032}}{\en{0.043}}{\en{2.34}}

\item لم يستطع أرهينيوس تفسير السلوك الحمضي لـ:
\opts{\chem{HCN}}{\chem{HClO}}{\chem{HI}}{\chem{NH4Cl}}

\item في التفاعل الآتي: \quad \chem{HSO4^-_{(aq)} + HCO3^-_{(aq)} <=> SO4^{2-}_{(aq)} + H2CO3_{(aq)}}
\par إذا علمت أن الاتزان ينزاح نحو تكوين المواد الناتجة، فإن الحمض الأقوى هو:
\opts{\chem{HSO4^-}}{\chem{HCO3^-}}{\chem{H2O}}{\chem{H2CO3}}

\item الحمض المرافق الناتج من تفاعل \chem{CH3COOH} مع \chem{(CH3)3N} هو:
\opts{\chem{CH3COO^-}}{\chem{CH3COOH}}{\chem{(CH3)3NH^+}}{\chem{H3O^+}}

\item أُذيب \en{0.8 g} من هيدروكسيد الصوديوم \chem{NaOH} في كمية من الماء حتى أصبح حجم المحلول \en{1 L}، ثم تفاعل \en{15 mL} منه تمامًا مع \en{20 mL} من حمض النتريك \chem{HNO3} مجهول التركيز، فإن تركيز الحمض \en{(M)} يساوي:
\en{($M_r$ NaOH = 40 g/mol)}
\opts{\en{0.1}}{\en{0.001}}{\en{0.02}}{\en{0.4}}

\item محلول حمض الهيدروكلوريك \chem{HCl} بتركيز \en{0.2 M}، يتفاعل \en{200 mL} منه تمامًا مع محلول القاعدة القوية \en{(X)}. فإذا كانت الكتلة المولية للقاعدة تساوي \en{56 g/mol}، فإن كتلة القاعدة \en{(X)} بالغرام تساوي:
\opts{\en{1.6}}{\en{3.56}}{\en{1.92}}{\en{2.24}}

\item محلول القاعدة الضعيفة \chem{C5H5N} بتركيز \en{0.01 M}، فإن:
\optsTwo{\en{pH = 12}}{\LTRfig{$[\mathrm{OH^-}] = 1\times10^{-2}\ \mathrm{M}$}}{\en{pOH < 2}}{\LTRfig{$[\mathrm{H_3O^+}] > 1\times10^{-12}\ \mathrm{M}$}}

\item الملح الذي يُعدّ ذوبانه في الماء تميّهًا هو:
\opts{\chem{KI}}{\chem{KBr}}{\chem{KF}}{\chem{KCl}}

\end{enumerate}

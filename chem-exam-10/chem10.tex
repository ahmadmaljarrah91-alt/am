\begin{enumerate}[label=\textbf{\arabic*-}, itemsep=0.2cm, leftmargin=*]
\item الخاصية المرتبطة بعدد الكم الفرعي \en{($l$)} هي:
\choices{الشكل العام للفلك}{معدل البعد عن النواة}{اتجاه الغزل}
\item يُعدّ العنصر انتقاليًا داخليًا إذا انتهى توزيعه الإلكتروني بالفلك:
\choices{\en{s}}{\en{p}}{\en{f}}
\item الذي يحدد الاتجاه الفراغي للفلك هو عدد الكم:
\choices{الرئيسي}{الفرعي}{المغناطيسي}
\item عدد الأفلاك الكلي في المستوى الرئيسي الثالث \en{($n = 3$)} هو:
\choices{3 أفلاك}{9 أفلاك}{18 فلكًا}
\item تشير المعادلة الآتية إلى: \qquad \ionEq
\choices{طاقة التأين للمغنيسيوم}{الألفة الإلكترونية للمغنيسيوم}{الكهروسلبية للمغنيسيوم}
\item واحدة من الصيغ الآتية تحتوي رابطة تساهمية ثلاثية:
\choices{\en{\ce{N2}}}{\en{\ce{H2}}}{\en{\ce{O2}}}
\item نوع الرابطة بين ذرات عنصر الصوديوم \en{\ce{Na}}:
\choices{فلزية}{أيونية}{تساهمية}
\item نوع الرابطة في مركب كلوريد البوتاسيوم \en{\ce{KCl}}:
\choices{أيونية}{فلزية}{تساهمية}
\item إحدى الآتية ليست من خصائص المركبات الأيونية:
\choices{موصلة للكهرباء وهي محلول}{متطايرة}{ذائبيتها عالية في الماء}
\item الرابطة في جزيء \en{\ce{CH4}} هي:
\choices{تساهمية}{فلزية}{أيونية}
\end{enumerate}

\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{|c|c|p{11.5cm}|}
\hline
\textbf{الفقرة} & \textbf{الإجابة} & \textbf{التوضيح} \\ \hline
1 & \textbf{أ} & الشكل العام للفلك (عدد الكم الفرعي يحدد نوع الفلك \en{s, p, d, f}). \\ \hline
2 & \textbf{جـ} & \en{f}، فالعناصر الانتقالية الداخلية (اللانثانيدات والأكتينيدات) ينتهي توزيعها بالفلك \en{f}. \\ \hline
3 & \textbf{جـ} & عدد الكم المغناطيسي يحدد الاتجاه الفراغي للفلك. \\ \hline
4 & \textbf{ب} & 9 أفلاك، لأن عدد الأفلاك $= n^2 = 3^2 = 9$. \\ \hline
5 & \textbf{أ} & طاقة التأين للمغنيسيوم (مجموع طاقتي التأين الأولى والثانية)، لأن الذرة تفقد إلكترونات بامتصاص طاقة. \\ \hline
6 & \textbf{أ} & \en{\ce{N2}}، فبين ذرتي النيتروجين رابطة ثلاثية \en{\ce{N#N}}. \\ \hline
7 & \textbf{أ} & فلزية، لأن الصوديوم فلز. \\ \hline
8 & \textbf{أ} & أيونية، بين فلز (\en{\ce{K}}) ولا فلز (\en{\ce{Cl}}). \\ \hline
9 & \textbf{ب} & متطايرة، فالمركبات الأيونية غير متطايرة لأن درجات غليانها مرتفعة. \\ \hline
10 & \textbf{أ} & تساهمية، بين لا فلزين (\en{\ce{C}} و \en{\ce{H}}). \\ \hline
\end{tabular}
